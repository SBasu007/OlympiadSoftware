document.addEventListener('DOMContentLoaded', function () {

    // Get the data from the hidden textarea
    const studentData = document.getElementById('studentData').value;

    // Parse JSON data
    const students = JSON.parse(studentData);

    let currentDocIndex = 0;

    // =========================
    // LOAD + PRINT FUNCTION
    // =========================
    function loadAndPrintDocument() {

        if (currentDocIndex < students.length) {

            // =========================
            // LOAD DATA INTO FIELDS
            // =========================

            document.getElementById('text-content1').textContent =
                students[currentDocIndex].name;

            document.getElementById('text-content2').textContent =
                students[currentDocIndex].guard_name;

            // Extract class
            let classValue;
            const subject = students[currentDocIndex].subject;

            if (/lower\s*nursery/i.test(subject)) {
                classValue = "Lower Nursery";
            }
            else if (/upper\s*nursery/i.test(subject)) {
                classValue = "Upper Nursery";
            }
            else if (/kg/i.test(subject)) {
                classValue = "KG";
            }
            else {
                const numMatch = subject.match(/(\d+)/);
                classValue = numMatch ? numMatch[0] : "N/A";
            }

            document.getElementById('text-content3').textContent =
                classValue;

            document.getElementById('text-content4').textContent =
                students[currentDocIndex].roll;

            document.getElementById('text-content5').textContent =
                students[currentDocIndex].school_name;

            // Subject name
            const subjectName = students[currentDocIndex].subject
                .replace(/[0-9]/g, "")
                .replace(/lower\s*nursery/gi, "")
                .replace(/upper\s*nursery/gi, "")
                .replace(/kg/gi, "")
                .trim();

            document.getElementById('text-content6').textContent =
                "Talent Search " +
                subjectName +
                " Olympiad, Class " +
                classValue;

            // District
            const district = students[currentDocIndex].district;

            const formattedDistrict =
                district.charAt(0).toUpperCase() +
                district.slice(1).toLowerCase();

            document.getElementById('text-content7').textContent =
                formattedDistrict + ", West Bengal";

            // Performance report
            document.getElementById('text-content8').textContent =
                "Student Performance Report (" +
                students[currentDocIndex].session +
                ")";

            document.getElementById('text-content9').textContent =
                "Total Marks - 100";

            let p = students[currentDocIndex].marks;

            document.getElementById('text-content10').textContent =
                "Marks Scored - " +
                students[currentDocIndex].marks;

            document.getElementById('text-content11').textContent =
                "Percentile Score - " +
                p +
                "%";

            // Date
            function postDate() {
                const d = new Date();

                let date =
                    d.getDate() +
                    "." +
                    (d.getMonth() + 1) +
                    "." +
                    d.getFullYear();

                return date;
            }

            document.getElementById('text-content12').textContent =
                "Class Rank - " +
                students[currentDocIndex].subject_school_rank;

            document.getElementById('text-content15').textContent =
                "District Rank - " +
                students[currentDocIndex].subject_school_rank;

            document.getElementById('text-content13').textContent =
                "State Rank - " +
                students[currentDocIndex].subject_overall_rank;

            document.getElementById('text-content14').textContent =
                "Date - " +
                postDate();

            // =========================
            // PRINT
            // =========================

            setTimeout(() => {
                window.print();
            }, 500);

        } else {

            alert("No more documents to print.");

        }
    }

    // =========================
    // AFTER PRINT
    // =========================
    window.onafterprint = () => {

        const proceed = confirm("Print next document?");

        if (proceed) {

            currentDocIndex++;

            if (currentDocIndex < students.length) {

                loadAndPrintDocument();

            } else {

                alert("No more documents to print.");

            }

        } else {

            alert("Printing stopped.");

        }
    };

    // =========================
    // START PRINTING
    // =========================
    window.onload = function () {
        loadAndPrintDocument();
    };

});
</script>
